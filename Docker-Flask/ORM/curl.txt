# Verificacion del servicio
curl http://localhost:5000/

# Registro de usuario
curl -X POST http://localhost:5000/register \
  -H "Content-Type: application/json" \
  -d '{"username": "julio", "password": "12345678"}'

# Inicio de sesion (devuelve access_token)
curl -X POST http://localhost:5000/login \
  -H "Content-Type: application/json" \
  -d '{"username": "julio", "password": "12345678"}'

# A partir de aqui, sustituye <TOKEN> por el access_token que devolvio /login

# Crear tarea
curl -X POST http://localhost:5000/tareas \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer <TOKEN>" \
  -d '{"titulo": "Estudiar para el examen", "descripcion": "Repasar temas 1-3"}'

# Listar tareas
curl http://localhost:5000/tareas \
  -H "Authorization: Bearer <TOKEN>"

# Actualizar tarea (id=1)
curl -X PUT http://localhost:5000/tareas/1 \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer <TOKEN>" \
  -d '{"completada": true}'

# Borrar tarea (id=1)
curl -X DELETE http://localhost:5000/tareas/1 \
  -H "Authorization: Bearer <TOKEN>"

# Probar sin token -> debe responder 401
curl http://localhost:5000/tareas
